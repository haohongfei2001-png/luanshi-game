extends SceneTree
const W := preload("res://scripts/world_sim.gd")
const H := preload("res://scripts/hometown_messages.gd")
const Driver := preload("res://tests/youth_combat_driver.gd")
const Setup := preload("res://scripts/hometown_persistence_smoke.gd")
func _initialize() -> void:
	var setup := Setup.new()
	var w := setup.build()
	print("MARCH ",w.youth_action("first_march"))
	var t := w.create_battle_ticket()
	var b := w.start_battle(t.ticket.battle_id)
	var outcome := Driver.play(b, "retreat")
	print("BATTLE ",outcome.result," dead=",outcome.forces[1].dead_ids.size()," elapsed=",outcome.elapsed_world_time)
	print("SETTLE ",w.commit_battle_outcome(outcome))
	print("ENDPOINT ",w.snapshot().youth.endpoint, " TIME ",w.snapshot().time_seconds)
	print("CHOICES ",w.youth_view().choices)
	var before := w.export_campaign_state()
	print("WAIT ",w.youth_action("wait_courier")," rolled_back=",before == w.export_campaign_state())
	var snapshot := w.snapshot()
	print("PRIOR TRUTH ",H.truth(snapshot))
	w._advance_clock_to(w.snapshot().time_seconds + 7200.0)
	print("LATER TRUTH ",H.truth(w.snapshot()))
	quit()
